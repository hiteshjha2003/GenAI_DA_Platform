# Importing necessary libraries
import streamlit as st  # Streamlit for creating the web app
import pandas as pd  # Pandas for data manipulation and analysis
from sqlalchemy import create_engine  # SQLAlchemy for database connection
import re  # Regular expressions for string operations
import matplotlib.pyplot as plt  # Matplotlib for data visualization


# Function to create a connection to the MS SQL Server
def create_connection(server, database):
    """
    Creates a connection to the MS SQL Server database.

    Parameters:
    server (str): The server name or IP address.
    database (str): The database name.

    Returns:
    engine: SQLAlchemy engine object for the database connection.
    """
    # Create a connection string with the given server and database using pyodbc driver
    connection_string = f"mssql+pyodbc://{server}/{database}?driver=ODBC+Driver+17+for+SQL+Server&trusted_connection=yes"
    # Create a connection engine
    engine = create_engine(connection_string)
    return engine

# Streamlit application Code
# Application title
st.title("DBA-EDA APP")

# Database Connections Parameters
server = st.text_input('Server')  # Input field for the server name
database = st.text_input('Database')  # Input field for the database name

# Create database connection
if st.button('Connect to Database', key='connect'):
    """
    Connect to the database when the button is pressed.
    """
    try:
        st.session_state.conn = create_connection(server, database)  # Create and store connection in session state
        st.success('Connected to Database successfully')  # Success message
    except Exception as e:
        st.error(f"Error connecting to database: {e}")  # Error message

# File Uploader
uploaded_file = st.file_uploader("Choose a file ", type=["csv", "xlsx"])

if uploaded_file is not None:
    """
    Load data from the uploaded file into a DataFrame.
    """
    if uploaded_file.name.endswith('.csv'):
        df = pd.read_csv(uploaded_file)  # Read CSV file
    elif uploaded_file.name.endswith('.xlsx'):
        df = pd.read_excel(uploaded_file)  # Read Excel file
    
    st.write('Uploaded Data')
    st.write(df)  # Display the uploaded data

    # Upload the dataframe into the database
    table_name = st.text_input('Table Name')  # Input field for table name
    if st.button('Upload to Database', key='upload'):
        """
        Upload the DataFrame to the database when the button is pressed.
        """
        if table_name:
            try:
                if 'conn' in st.session_state:  # Check if connection exists
                    conn = st.session_state.conn 
                    with conn.begin() as transaction:
                        df.to_sql(table_name, conn, if_exists='replace', index=False)  # Upload DataFrame to the specified table
                        transaction.commit()
                    st.success(f"Table {table_name} created successfully!")
                else:
                    st.error("Please connect to the database first")
            except Exception as e:
                st.error(f"Error uploading the file to the database: {e}")

    # Button to calculate the null, missing values, and duplicated values
    if st.button('Calculate Null, Missing Values', key='calculate'):
        """
        Calculate null, missing, and duplicated values in the DataFrame.
        """
        null_values = df.isnull().sum()  # Calculate null values
        duplicated_values = df.duplicated().sum()  # Calculate duplicated values
        missing_values = df.isnull().count()  # Calculate missing values
        st.write(f"Total null values: {null_values}")
        st.write(f"Total Missing Values: {missing_values}")
        st.write(f"Total Duplicated Values: {duplicated_values}")

    # Button to remove the null, missing values, and duplicated values
    if st.button('Remove Null, Missing Values', key='remove'):
        """
        Remove null, missing, and duplicated values from the DataFrame.
        """
        df_cleaned = df.dropna()  # Drop rows with null values
        st.write("Data is cleaned")
        st.write(df_cleaned)  # Display cleaned data

        # Update the cleaned dataframe in the database
        if table_name:
            try:
                if 'conn' in st.session_state:  # Check if connection exists
                    conn = st.session_state.conn
                    with conn.begin() as transaction:
                        df_cleaned.to_sql(table_name, conn, if_exists='replace', index=False)  # Upload cleaned DataFrame
                    st.success(f"Table {table_name} updated with cleaned data successfully")
                else:
                    st.error("Please connect to the database first")
            except Exception as e:
                st.error(f"Error uploading the file to the database: {e}")

# SQL Query Input
query = st.text_area('Enter your SQL Query:')  # Text area for SQL query input

if st.button('Execute Query', key='execute'):
    """
    Execute the SQL query and display the results.
    """
    if 'conn' in st.session_state:
        try:
            query_result = pd.read_sql_query(query, st.session_state.conn)  # Execute the query
            st.write('Query Results')
            st.write(query_result)  # Display the query results
        except Exception as e:
            st.error(f"Error executing query: {e}")
    else:
        st.error("Please connect to the database first")

# List Tables
if st.button('List Tables', key='list'):
    """
    List all tables in the connected database.
    """
    if 'conn' in st.session_state:
        try:
            query_result = pd.read_sql_query("SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_TYPE = 'BASE TABLE'", st.session_state.conn)
            st.write('Tables in the Database:')
            st.write(query_result)  # Display the list of tables
        except Exception as e:
            st.error(f"Error listing tables: {e}")

# Visualization
if 'conn' in st.session_state:
    tables_query = "SELECT TABLE_NAME FROM INFORMATION_SCHEMA.TABLES WHERE TABLE_TYPE = 'BASE TABLE'"
    tables = pd.read_sql_query(tables_query, st.session_state.conn)  # Query to get table names
    table_name = st.selectbox('Select Table', tables['TABLE_NAME'])  # Dropdown to select table

    col1, col2 = st.columns(2)
    with col1:
        x_col = st.text_input('X-axis Column')  # Input for X-axis column name
    with col2:
        y_col = st.text_input('Y-axis Column')  # Input for Y-axis column name

    chart_type = st.selectbox('Select Chart Type', ['Bar Chart', 'Scatter Chart', 'Bubble Chart', 'Histogram'])  # Dropdown to select chart type
    nums_rows = st.number_input('Number of Rows', min_value=1, value=100)  # Number input for number of rows to fetch

    if st.button('Generate Chart', key='chart'):
        """
        Generate the selected chart type using the specified columns and number of rows.
        """
        if x_col and y_col and chart_type:
            if table_name:
                query = f"SELECT TOP {nums_rows} [{x_col}], [{y_col}] FROM [{table_name}]"  # SQL query to fetch data
                data = pd.read_sql_query(query, st.session_state.conn)

                if chart_type == 'Bar Chart':
                    st.bar_chart(data.set_index(x_col))  # Generate bar chart
                elif chart_type == 'Scatter Chart':
                    fig, ax = plt.subplots()
                    data.plot.scatter(x=x_col, y=y_col, ax=ax)  # Generate scatter chart
                    st.pyplot(fig)
                elif chart_type == 'Bubble Chart':
                    fig, ax = plt.subplots()
                    data.plot.scatter(x=x_col, y=y_col, s=data[y_col]*10, ax=ax)  # Generate bubble chart
                    st.pyplot(fig)
                elif chart_type == 'Histogram':
                    fig, ax = plt.subplots()
                    data[x_col].plot.hist(ax=ax)  # Generate histogram
                    st.pyplot(fig)
            else:
                st.error('Select a table first')
        else:
            st.error('Please input both column names and select chart types')
else:
    st.error('Connect to a database first')
